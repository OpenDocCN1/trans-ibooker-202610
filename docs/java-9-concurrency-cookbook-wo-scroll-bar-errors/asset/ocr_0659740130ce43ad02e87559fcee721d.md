```
@Override
public void run() {
    for (int i=0; i<array.length; i++) {
        array[i]++;
        try {
            TimeUnit.MILLISECONDS.sleep(1);
        } catch (InterruptedException e) {
            e.printStackTrace();
        }
    }
}
```
