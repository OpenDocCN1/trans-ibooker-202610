    """对多项式取负，返回一个新的多项式。

    返回
    -------
    :class:`UniPoly`
        一个新的多项式。
    """
    # 逐项取负多项式，除非它已经是 0
    if not self.__terms:
        return self

    new_poly = UniPoly(0, self.__indeterminate)
    for exponent in self.__terms:
