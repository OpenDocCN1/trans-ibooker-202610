pub fn run(config: Config) -> MyResult<()> {
    let mut file = open(&config.in_file)
        .map_err(|e| format!("{}: {}", config.in_file,
e))?; ①
    let mut line = String::new(); ②
    loop { ③
        let bytes = file.read_line(&mut line)?; ④
        if bytes == 0 { ⑤
            break;
        }
        print!("{}", line); ⑥
        line.clear(); ⑦
    }
    Ok(())
}
