fn run_stdin(
    input_file: &str, ①
    args: &[&str],
    expected_file: &str,
) -> TestResult {
    let input = fs::read_to_string(input_file)?; ②
    let expected = fs::read_to_string(expected_file)?;
    Command::cargo_bin(PRG)? ③
        .args(args)
        .write_stdin(input)

        .success()
        .stdout(expected);

}
