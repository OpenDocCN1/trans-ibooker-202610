6. In the private section, define the fTime function, which returns std::string. It gets a system time. Format it into a readable representation and return it:

```
std::string fTime()
{
    auto now = std::chrono::system_clock::now();
    std::time_t time = std::chrono::system_clock::to_time_t(now);
    std::ostringstream oss;
    std::string strTime(std::ctime(&time));
    strTime.pop_back();
    oss << "[" << strTime << "]";
    return oss.str();
}
```

7. In the private section, define the threadId() function, which returns a string. Get the id of the current thread, format it, and return it:

```
std::string threadId()
{
    auto id = std::this_thread::get_id();
    std::ostringstream oss;
    oss << "[" << std::dec << id << "]";
    return oss.str();
}
```
