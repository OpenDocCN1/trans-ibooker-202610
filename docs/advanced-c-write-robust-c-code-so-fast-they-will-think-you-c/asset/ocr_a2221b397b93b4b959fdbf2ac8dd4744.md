+   4. Create a function named **fun1()** that just loops over each string in fruits, copies it to a string, and sums the characters of that string:

```
```cpp
uint64_t fun1()
{
    TIME_IT;
    uint64_t sum = 0;
    string s1;
    for(uint64_t i = 0; i < N; ++i)
    {
        s1 = fruits[i % 10];
        for(int k = 0; k < s1.size(); ++k) sum += s1[k];
    }
    return sum;
}

```cpp
uint64_t fun2()
{
    TIME_IT;
    uint64_t sum = 0;
    char s1[32];

    for(uint64_t i = 0; i < N; ++i)
    {
        char *ps1 = s1;
        const char *p1 = fruits[i % 10];
        do { *ps1++ = *p1; } while(*p1++);
        for(ps1 = s1, *ps1; ++ps1) sum += *ps1;
    }
    return sum;
}
