For the demo, we build a program to evaluate the num_a value with a switch statement. We set a random value with a maximum value of 5. Open Arduino software and write this complete program.

```
```cpp
long num_a;

void setup() {
    // initialize serial communication
    Serial.begin(115200);
    while (!Serial) {
        ;
    }
}
void loop() {
    num_a = random(5);

    // print
    Serial.print("num_a: ");
    Serial.println(num_a);
    switch(num_a) {
        case 0:
            Serial.println("num_a value is 0");
            break;
        case 1:
            Serial.println("num_a value is 1");
            break;
        case 2:
            Serial.println("num_a value is 2");
            break;
