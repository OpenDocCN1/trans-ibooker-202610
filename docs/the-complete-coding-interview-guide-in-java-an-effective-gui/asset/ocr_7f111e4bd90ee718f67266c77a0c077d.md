public MyStackViaQueue() {
    queue1 = new ArrayDeque<>();
    queue2 = new ArrayDeque<>();
}

public void push(E e) {
    if (!queue1.isEmpty()) {
        if (peek != null) {
            queue1.add(peek);
        }
        queue1.add(e);
    } else {
        if (peek != null) {
            queue2.add(peek);
        }
        queue2.add(e);
    }
    size++;
    peek = null;
}

public E pop() {
    if (size() == 0) {
        throw new EmptyStackException();
    }
    if (peek != null) {
        E e = peek;
        peek = null;
        size--;
        return e;
    }
}
