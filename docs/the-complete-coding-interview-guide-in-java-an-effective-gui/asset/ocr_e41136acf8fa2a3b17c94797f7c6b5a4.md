public static int maxProfitOneTransaction(int[] prices) {

    int min = prices[0];
    int result = 0;
    for (int i = 1; i < prices.length; i++) {
        result = Math.max(result, prices[i] - min);
        min = Math.min(min, prices[i]);
    }

    return result;
}
