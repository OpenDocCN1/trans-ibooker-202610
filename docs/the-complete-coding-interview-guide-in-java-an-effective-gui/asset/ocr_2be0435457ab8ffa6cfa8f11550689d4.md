```java
private static void merge(int[] result,
	int[] left, int[] right) {
	int t1 = 0;
	int t2 = 0;
	for (int i = 0; i < result.length; i++) {
		if (t2 >= right.length
		|| (t1 < left.length && left[t1] <= right[t2])) {
			result[i] = left[t1];
			t1++;
		} else {
			result[i] = right[t2];
			t2++;
		}
	}
}
