public static boolean checkDuplicates(int[] arr) {
    for (int i = 0; i < arr.length; i++) {
        if (arr[Math.abs(arr[i])] > 0) {
            arr[Math.abs(arr[i])] = -arr[Math.abs(arr[i])];
        } else if (arr[Math.abs(arr[i])] == 0) {
            arr[Math.abs(arr[i])] = -(arr.length-1);
        } else {
            return true;
        }
    }
