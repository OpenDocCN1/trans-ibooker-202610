    conn = sqlite3.connect(DATABASE)
    c = conn.cursor()
    c.execute(
        "CREATE TABLE IF NOT EXISTS keywords (keyword TEXT, score REAL, hashtags TEXT)"
    )
    for keyword, score, hashtags in zip(keywords, score, hashtags):
        c.execute("INSERT INTO keywords VALUES (?, ?, ?)", (keyword, score, hashtags))
    conn.commit()
    conn.close()
    return db_exists

def collect_extract(filename):
    keywords = []
    score = []
    text = read_file(filename)
    extracted_keyword_score = extract_keywords(text)
    for keyword_score in extracted_keyword_score:
        keywords.append(keyword_score[0])
        score.append(keyword_score[1])
    hashtags = make_hashtags(extracted_keyword_score)
    return keywords, score, hashtags

def extract_and_load(filename):
    keywords, score, hashtags = collect_extract(filename)

#!/usr/bin/env python
"""Create Zero-shot classification command-line tool using Hugging Face's transformers library."""

from transformers import pipeline
import click

def read_file(filename):
    with open(filename, encoding="utf-8") as myfile:
        return myfile.read()

def read_labels(kw_file):
    return read_file(kw_file).splitlines()

def classify(text, labels, model="MoritzLaurer/mDeBERTa-v3-base-mnli-xnli"):
    classifier = pipeline("zero-shot-classification", model=model)
    results = classifier(text, labels, multi_label=False)
    return results

@click.group()
def cli():
    """A cli for zero-shot classification"""

@cli.command("classify")
@click.argument("filename", default="four-score.m4a.txt")
@click.argument("kw_file", default="keywords.txt")
def classifycli(filename, kw_file):
    """Classify text using zero-shot classification"""
    text = read_file(filename)
    labels = read_labels(kw_file)
    results = classify(text, labels)
    for label, score in zip(results["labels"], results["scores"]):
        click.secho(f"{label}\t{score:.2f}", fg="green")

if __name__ == "__main__":
    cli()

# ML 系统中的数据投毒预防

数据投毒的概念，即污染训练数据，是每个从事 MLOps 的组织都应放在首要考虑的问题之一。即使是由全球最优秀的从业者设计的系统，也可能出现令人意外的非预期后果^([1](#ch09.xhtml#idm46247689795440))。

开始讨论此场景的一种方式是 [图 9-12](#ch09.xhtml#Figure-10-3-data-poisoning)，它展示了两种潜在的攻击向量。在第一个攻击向量中，拥有类似 Google Drive 或 iCloud 或任何云存储系统写入权限的员工或承包商，该系统会筛查禁止的图片，可能故意植入禁止的图片或触发禁止图片误报的图片。在第二次攻击中，使用 AI 扫描禁止图片的公司的内部人员通过足够多的误报污染训练数据集，从而有可能欺骗机器学习系统日后标记具有相同误报的账户。

# 推荐引擎

推荐引擎是 MLOps 中一个令人兴奋问题的核心^([2](#ch09.xhtml#idm46247689750256))。一方面，推荐引擎对许多组织而言是“基本门槛”，也是现代机器学习管道中最古老的问题之一。最著名的例子是 Netflix 大奖。最著名的是，它并未 [在生产环境中实现](https://netflixtechblog.com/netflix-recommendations-beyond-the-5-stars-part-1-55838468f429)，即早在 2012 年就出现的一个原始 MLOps 问题。故事的大致内容是，最初 Netflix 在竞赛中优化的是准确性，而非运营复杂性，而我们现在称之为 MLOps。面对“获胜”方案的难度，他们放弃了它，转而采用更易于部署到生产环境的东西。

关键问题在于工程要求过于复杂，且问题框架发生了变化。这些是超越推荐引擎类别的关键问题，对任何 MLOps 问题都至关重要。尽管如此，MLOps 与推荐引擎相关的一些独特问题也值得讨论。让我们深入探讨用户体验、负外部性和启发式方法。

## 推荐引擎用户体验

推荐引擎最重要的一部分可能并非推荐引擎本身，而是围绕它的用户体验。让我们快速看一个使用 [GitHub](https://github.com) 的例子。当本章于 2022 年撰写时，Github Feed 在我的 GitHub 动态中推荐用户关注。虽然许多人喜欢深入研究推荐引擎中涉及的机器学习算法的复杂性，但一个关键的 MLOps 思维问题是用户体验。在 [图 9-13](#ch09.xhtml#Figure-10-1-recommendation) 中，Noah 的 GitHub 动态显示了两个推荐用户（用户的姓名已被涂黑以保护隐私）。不过，如果 Noah 想避免关注他们，就会变得有趣起来，也许是因为他不喜欢他们的代码，他已经关注太多人，或出于其他各种原因。
