function _fd_write(fd, iov, iovcnt, pnum) {
  // hack to support printf in SYSCALLS_REQUIRE_FILESYSTEM=0
  var num = 0;
  for (var i = 0; i < iovcnt; i++) {
    var ptr = HEAP32[(((iov)+(i*8))>>2)];
    var len = HEAP32[(((iov)+(i*8 + 4))>>2)];
    for (var j = 0; j < len; j++) {
      SYSCALLS.printChar(fd, HEAPU8[ptr+j]);
    }
    num += len;
  }
  HEAP32[((pnum)>>2)] = num
  return 0;
