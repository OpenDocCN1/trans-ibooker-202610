def get_order():

        return {}

    drinks = read_menu("drinks.txt")
    flavors = read_menu("flavors.txt")
    toppings = read_menu("toppings.txt")
    order["drink"] = menu(drinks, "Erik's drinks",

                           "Choose your flavor: ")
    order["topping"] = menu(toppings, "Erik's toppings",
                            "Choose your topping: ")
