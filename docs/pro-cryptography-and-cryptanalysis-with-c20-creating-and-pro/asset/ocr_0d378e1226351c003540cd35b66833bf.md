//** a standard header file of GNU C++ library.

#include<stdio.h>

using namespace std;

//** the function will compute
//** the GCD for two number
int g(int x, int y) {
    if (x == 0)
        return y;
    return g(y % x, x);
}

int main()
{
    int x = 10, y = 15;
    cout << "Euclid GCD(" << x << ", "
         << y << ") = " << g(x, y)
