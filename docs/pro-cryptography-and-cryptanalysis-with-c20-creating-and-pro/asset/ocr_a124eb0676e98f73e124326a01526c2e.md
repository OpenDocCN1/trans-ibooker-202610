// Auxiliary function that prints a matrix on the console
void print_matrix(double matrix[100][100], int n, string message)
{
    cout<<endl<<"***"<<message<<"***"<<endl;

    for(int i = 0; i < n; i++)
    {
        for(int j = 0; j < n; j++)
            cout<<matrix[i][j]<<"    ";

        cout<<endl;
    }

    cout<<endl;
}
