class NoteListItemComponent extends HTMLElement {

    constructor() {
        super();

        this._note = JSON.parse(this.getAttribute('note')) || {};
        this.idx = this.getAttribute('idx') || -1;
        this.root = this.attachShadow({mode: 'open'});
        this.root.innerHTML = this.getTemplate();
    }

    get note() {
        return this._note;
    }

    set note(newValue) {
        this._note = newValue;
    }

    get idx() {
        return this._idx;
    }
}
