
```
void setup() {
  // Set up logging. Google style is to avoid globals or statics because of
  // lifetime uncertainty, but since this has a trivial destructor it's okay.
  static tflite::MicroErrorReporter micro_error_reporter; //NOLINT
  error_reporter = &micro_error_reporter;

  // Map the model into a usable data structure. This doesn't involve any
  // copying or parsing, it's a very lightweight operation.
  model = tflite::GetModel(g_magic_wand_model_data);
  if (model->version() != TFLITE_SCHEMA_VERSION) {
    error_reporter->Report(
        "Model provided is schema version %d not equal "
        "to supported version %d.",
        model->version(), TFLITE_SCHEMA_VERSION);
    return;
  }

  // Pull in only the operation implementations we need.
  // This relies on a complete list of all the ops needed by this graph.
  // An easier approach is to just use the AllOpsResolver, but this will
  // incur some penalty in code space for op implementations that are not
  // needed by this graph.
  static tflite::MicroMutableOpResolver micro_mutable_op_resolver; // NOLINT
  micro_mutable_op_resolver.AddBuiltin(
      tflite::BuiltinOperator_DEPTHWISE_CONV_2D,
      tflite::ops::micro::Register_DEPTHWISE_CONV_2D());
  micro_mutable_op_resolver.AddBuiltin(
      tflite::BuiltinOperator_MAX_POOL_2D,
      tflite::ops::micro::Register_MAX_POOL_2D());
  micro_mutable_op_resolver.AddBuiltin(
      tflite::BuiltinOperator_CONV_2D,
      tflite::ops::micro::Register_CONV_2D());
  micro_mutable_op_resolver.AddBuiltin(
      tflite::BuiltinOperator_FULLY_CONNECTED,
      tflite::ops::micro::Register_FULLY_CONNECTED());
  micro_mutable_op_resolver.AddBuiltin(tflite::BuiltinOperator_SOFTMAX,
                                       tflite::ops::micro::Register_SOFTMAX());

  // Build an interpreter to run the model with
  static tflite::MicroInterpreter static_interpreter(model,
                                                     micro_mutable_op_resolver,
                                                     tensor_arena,
                                                     kTensorArenaSize,
                                                     error_reporter);
  interpreter = &static_interpreter;

  // Allocate memory from the tensor_arena for the model's tensors
  interpreter->AllocateTensors();

  // Obtain pointer to the model's input tensor
  model_input = interpreter->input(0);
  if ((model_input->dims->size != 4) || (model_input->dims->data[0] != 1) ||
      (model_input->dims->data[1] != 128) ||
      (model_input->dims->data[2] != kChannelNumber) ||
      (model_input->type != kTfLiteFloat32)) {
    error_reporter->Report("Bad input tensor parameters in model");
    return;
  }

  input_length = model_input->bytes / sizeof(float);

  TfLiteStatus setup_status = SetupAccelerometer(error_reporter);
  if (setup_status != kTfLiteOk) {
    error_reporter->Report("Set up failed\n");
  }
}
```

より興味深いことは、`loop()` 関数で起こる：

```
void loop() {
  // Attempt to read new data from the accelerometer
  bool got_data = ReadAccelerometer(error_reporter, model_input->data.f,
                                    input_length, should_clear_buffer);
  // Don't try to clear the buffer again
  should_clear_buffer = false;
  // If there was no new data, wait until next time
  if (!got_data) return;
  // Run inference, and report any error
  TfLiteStatus invoke_status = interpreter->Invoke();
  if (invoke_status != kTfLiteOk) {
    error_reporter->Report("Invoke failed on index: %d\n", begin_index);
    return;
  }
  // Analyze the results to obtain a prediction
  int gesture_index = PredictGesture(interpreter->output(0)->data.f);
  // Clear the buffer next time we read data
  should_clear_buffer = gesture_index < 3;
  // Produce an output
  HandleOutput(error_reporter, gesture_index);
}
```

まず、加速度センサーから値を読み取ろうとする。試行後、`should_clear_buffer` を`false` にセットし、当分の間、クリアしようとするのを止める。

新しいデータの取得に失敗した場合、`ReadAccelerometer()` は`false` の値を返し、次に呼び出されたときに再挑戦できるように`loop()` 関数からリターンする。

`ReadAccelerometer()` が返す値が`true` の場合、入力されたばかりの入力テンソルで推論を実行する。その結果を`PredictGesture()` に渡すと、どのジェスチャーが検出されたかのインデックスが得られる。インデックスが`3` より小さければ、そのジェスチャは有効なので、次に`ReadAccelerometer()` が呼び出されたときにバッファをクリアするために`should_clear_buffer` フラグをセットする。その後、`HandleOutput()` を呼び出して、ユーザに結果を報告する。

`main.cc`では、`main()` 関数がプログラムをキックオフし、`setup()` 関数を実行し、`loop()` 関数をループ呼び出ししている：

```
int main(int argc, char* argv[]) {
  setup();
  while (true) {
    loop();
  }
}
```

# マイクロコントローラへのデプロイ

このセクションでは、コードを2つのデバイスにデプロイする：

*   [Arduino Nano 33 BLEセンス](https://oreil.ly/6qlMD)

*   [スパークファン・エッジ](https://oreil.ly/-hoL-)

Arduinoの実装から始めよう。

## Arduino

Arduino Nano 33 BLE Senseは、3軸加速度センサーとBluetoothをサポートしており、魔法の杖を作るのに最適な小型・軽量だ。

アプリケーションの主要ファイルのArduino固有の実装を見ていこう。

### Arduino定数

定数`kConsecutiveInferenceThresholds` は、[`arduino/constants.cc`](https://oreil.ly/5bBt0)ファイルで再定義されている：

```
// The number of expected consecutive inferences for each gesture type.
// Established with the Arduino Nano 33 BLE Sense.
const int kConsecutiveInferenceThresholds[3] = {8, 5, 4};
```

この章で前述したように、この定数には、各ジェスチャーが検出されたとみなされるために必要な、連続した正の推論の数が格納される。この数は1秒間に実行される推論の数に依存し、デバイスによって異なる。デフォルトの数値はSparkFun Edge用に調整されているため、Arduino実装には独自の数値セットが必要だ。これらのしきい値を変更することで、推論をより難しくしたり、トリガーしやすくしたりすることができるが、低く設定しすぎると誤検出の原因になる。

### Arduinoで加速度センサーのデータを取得する

Arduino加速度センサー・ハンドラーは[`arduino/accelerometer_handler.cc`](https://oreil.ly/jV_Qm)にある。加速度センサーからデータを取り込み、モデルの入力バッファに書き込む。

今回使用するモデルは、スパークファン・エッジ・ボードのデータを使ってトレーニングした。Edgeの加速度センサーは、25Hz、つまり1秒間に25回の読み取りをセットで提供する。正しく動作するためには、同じレートでキャプチャされたデータを供給する必要がある。結局のところ、Arduino Nano 33 BLE Senseボードの加速度センサーは119Hzのレートで測定値を返す。つまり、データをキャプチャするだけでなく、モデルに合わせて*ダウンサンプリング*する必要がある。

とても専門的に聞こえるが、ダウンサンプリングは実はとても簡単だ。信号のサンプル・レートを下げるには、データの一部を捨てればいい。次のコードで、この仕組みを見てみよう。

まず実装には、それ自身のヘッダファイルといくつかの他のファイルが含まれている：

```
#include "tensorflow/lite/micro/examples/magic_wand/
  accelerometer_handler.h"

#include <Arduino.h>
#include <Arduino_LSM9DS1.h>

#include "tensorflow/lite/micro/examples/magic_wand/constants.h"
```

`Arduino.h`ファイルは、Arduinoプラットフォームの基本的な機能へのアクセスを提供する。`Arduino_LSM9DS1.h`は [Arduino_LSM9DS1](https://oreil.ly/eb3Zs)ライブラリの一部で、ボードの加速度センサーと通信するために使用する。

次に、いくつかの変数をセットする：

```
// A buffer holding the last 200 sets of 3-channel values
float save_data[600] = {0.0};
// Most recent position in the save_data buffer
int begin_index = 0;
// True if there is not yet enough data to run inference
bool pending_initial_data = true;
// How often we should save a measurement during downsampling
int sample_every_n;
// The number of measurements since we last saved one
int sample_skip_counter = 1;
```

これらの変数には、データを入れるバッファ、`save_data` 、バッファ内の現在位置と推論を開始するのに十分なデータがあるかどうかを追跡するための変数が含まれる。最も興味深い2つの変数、`sample_every_n` と`sample_skip_counter` は、ダウンサンプリングプロセスで使用される。これについてはもう少し詳しく見ていこう。

ファイルの次に、`SetupAccelerometer()` 関数がプログラムのメインループによって呼び出され、ボードをデータ・キャプチャーの準備に入れる：

```
TfLiteStatus SetupAccelerometer(tflite::ErrorReporter* error_reporter) {
  // Wait until we know the serial port is ready
  while (!Serial) {
  }

  // Switch on the IMU
  if (!IMU.begin()) {
    error_reporter->Report("Failed to initialize IMU");
    return kTfLiteError;
  }
```

すべての準備が整ったことを示すメッセージを出力するため、最初に行うことは、デバイスのシリアルポートの準備が整っていることを確認することだ。それから*慣性計測ユニット*（IMU）のスイッチを入れる。IMUは加速度センサーを含む電子コンポーネントである。`IMU` オブジェクトは、`Arduino_LSM9DS1` ライブラリから来ている。

次のステップは、ダウンサンプリングについて考え始めることだ。まずIMUライブラリにクエリを発行し、ボードのサンプルレートを決定する。その数値がわかったら、`kTargetHz` 、[`constants.h`](https://oreil.ly/rQaSw)の一部として定義されているターゲット・サンプル・レートで割る：
