void add(String added) {
  if(a.isEmpty) {
    if(!regExp.hasMatch(added)) {
      a.add(added);
    }
  } else if(regExp.hasMatch(a.last)) {
    if(!RegExp("[+|-x÷/]").hasMatch(added)) {
      a.add(added);
    }
  } else {
    if(regExp.hasMatch(added)) {
      if(!RegExp(\".\").hasMatch(a.last)) a.last+=".0";
      a.add(added);
    } else {
      a.last+=added;
    }
  }
}
