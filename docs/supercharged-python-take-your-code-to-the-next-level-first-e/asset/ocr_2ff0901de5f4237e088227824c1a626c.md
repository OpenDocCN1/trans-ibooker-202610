import time

def gen_rand():
    p1 = 1200556037
    p2 = 2444555677
    max_rand = 2 ** 32
    r = int(time.time() * 1000)
    while True:
        n = r
        n *= p2
        n %= p1
        n += r
        n *= p1
        n %= p2
        n %= max_rand
        r = n
        yield n
