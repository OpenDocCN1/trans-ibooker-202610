suits = ['clubs', 'diamonds', 'hearts', 'spades' ]
my_deck = Deck(52)

# Deal twelve poker hands, so user can compare before
# and after shuffling.

for i in range(12):
    for i in range(5):
        d = my_deck.deal()
        r = d % 13
        s = d // 13
        print(ranks[r], 'of', suits[s])
    print()
