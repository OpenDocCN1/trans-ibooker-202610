class Dog:
    def __init__(self, d):
        self.d = d

    def __gt__(self, other):
        """ Greater than (>). This method provides a
        less-than comparison through Rules of Symmetry.
        if a > b, then b < a.
        """
        if type(other) == Dog:
            return self.d > other.d
        else:
            return self.d > other

    def __lt__(self, other):
        """ Less than (<). This method must support
        comparisons to objects of the same class, as well
        as to numbers.
        """
        if type(other) == Dog:
            return self.d < other.d
        else:
            return self.d < other

# Defining __repr__ also gets us __str__.
def __repr__(self):
    return "Dog(" + str(self.d) + ")"
