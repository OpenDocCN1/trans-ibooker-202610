def load_stock(ticker_str):
    """ Load stock function.

    Given a short string, ticker_str, load information
    for the indicated stock, such as 'MSFT,' into a Pandas
    data frame (df) and return it.
    """
    df = web.DataReader(ticker_str, 'yahoo')
    df = df.reset_index()
    return df
