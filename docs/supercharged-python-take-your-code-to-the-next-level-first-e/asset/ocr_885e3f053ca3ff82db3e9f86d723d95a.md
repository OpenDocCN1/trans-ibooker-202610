def get_str(s):
    """Get String helper function.
    Get the quoted portion of a string by getting
    text
    from the first quote mark to the last quote
    mark. If
    these aren't present, return an empty string.
    """
    a = s.find(""")
    b = s.rfind(""")
    if a == -1 or b == -1:
        return ''
    return s[a+1:b]
