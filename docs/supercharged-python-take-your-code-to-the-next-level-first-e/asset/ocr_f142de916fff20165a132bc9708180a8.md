from fractions import Fraction
total = Fraction('0')
while True:
    s = input('Enter fraction (press ENTER to quit): ')
    s = s.replace(' ', '') # Elim. spaces, just in case.
    if not s:
        break
    total += Fraction(s)
print('The total is %s.' % total)
