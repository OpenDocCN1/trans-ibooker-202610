import numpy as np
import matplotlib.pyplot as plt

A = np.random.standard_normal(2000000)
A = A * 10 + 100
B, X = np.histogram(A, 50)
X = (X[1:]+X[:-1])/2   # Use bin centers rather than edges.
plt.plot(X, B)         # Plot against values rather than
plt.show()             #   bin numbers.
