    '''Jump on Not Zero operation.
    After evaluating the operands, test the first
    op;
    if not zero, set Program Counter to op2 - 1.
    '''
    global pc
    op2, op1 = stack.pop(), stack.pop()

        op1 = sym_tab[op1]
    if type(op2) == str:
        op2 = sym_tab[op2]
    if op1:
        pc = int(op2) - 1    # Convert op to int
