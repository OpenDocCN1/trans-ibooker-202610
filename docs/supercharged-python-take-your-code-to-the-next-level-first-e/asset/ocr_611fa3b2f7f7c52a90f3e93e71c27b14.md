import random
n = random.randint(1, 50)
while True:
    guess = int(input('Enter guess:'))
    if guess == n:
        print('Success! You win.')
        break
    elif guess < n:
        print('Too low.', end=' ')
    else:
        print('Too high.', end=' ')
