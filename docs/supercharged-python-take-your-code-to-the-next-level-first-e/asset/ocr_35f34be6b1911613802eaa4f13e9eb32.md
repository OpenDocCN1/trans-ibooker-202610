    """Open-source-file function. Open a named
    file and read lines into a list, which is
    returned.
    """
    while True:
        try:
            fname = input('Enter RPN source: ')
            f = open(fname, 'r')
            if not f:
                return None
            else:

        except:
            print('File not found. Re-enter.')
