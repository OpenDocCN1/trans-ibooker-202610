# 8.3.4. Exceptions in C++

In C++, using example 8.5, a warning is generated during compilation, but the resulting program can be executed and behaves in the same way as it did in Python, as we see from example 8.7.

EXAMPLE 8.7.--

```
```cpp
#include <iostream>
#include <exception>
using namespace std ;

class MyExc : public exception {
public :
virtual const char* what () const throw () { return ("MyExc") ; }
} ;

class MyDerExc : public MyExc {
public :
virtual const char* what () const throw () { return ("MyDerExc") ; }
} ;

class MyExc exc ;
class MyDerExc dec ;

int main () {
try { throw exc ; }
catch (exception& e) { cout << e.what () << '\n' ; }
try { throw dec ; }
catch (exception& e) { cout << e.what () << '\n' ; }
try { throw dec ; }
catch (MyDerExc& e) { cout << "MyDerExc " << e.what () << '\n' ; }
catch (MyExc& e) { cout << "MyExc " << e.what () << '\n' ; }
try { throw dec ; }
catch (MyExc& e) { cout << "MyExc " << e.what () << '\n' ; }
catch (MyDerExc& e) { cout << "MyDerExc"<< e.what () << '\n' ; }
return (0) ;
}
```
