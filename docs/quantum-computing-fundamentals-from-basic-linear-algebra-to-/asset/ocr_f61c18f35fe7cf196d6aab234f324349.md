operation RandomNumberInRange(max : Int) : Int {
    mutable bits = new Result[0];
    for (idxBit in 1..BitsToSize(max)) {
        set bits += [QuantumPseudoRandomNumberGenerator()];
    }
    let sample = ResultArrayAsInt(bits);
    return sample > max ?
        RandomNumberInRange(max)
    |
        sample;
}
