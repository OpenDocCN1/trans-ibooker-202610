_logger = logger;
}

As you can see, we're passing the logger in; therefore, we're instantiating it in the Main method:

static void Main(string[] args)
{
    var path = args[0];
    var logger = new FileLogger(path);

    using var service = new PhotoService(path, logger);
    ServiceBase.Run(service);
}

This has had a little re-jig: now, we're taking the argument and storing it in the path variable.
