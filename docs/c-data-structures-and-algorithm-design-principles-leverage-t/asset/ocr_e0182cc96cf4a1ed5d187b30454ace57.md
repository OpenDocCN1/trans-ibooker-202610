9. From our main() function, we can now output the results of each strongly connected component by printing the values of each vector on a separate line:

```
int main()
{
    int V;
    vector<vector<int>> adj;

    auto connectedComponents = Kosaraju(V, adj);

    cout << "Graph contains " << connectedComponents.size() << " strongly connected components." << endl;

    for(auto component : connectedComponents)
    {
        cout << "\t";

        for(auto node : component)
        {
            cout << node << " ";
        }
        cout << endl;
    }
}
```
