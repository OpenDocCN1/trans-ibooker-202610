def run_host(ip)
  begin
    connect
    res = send_request_raw({ 'uri' => '/',
      'method' => 'GET' })
    fp = http_fingerprint(:response => res)
    print_good("#{ip}:#{rport} #{fp}") if fp
    report_service(:host => rhost, :port => rport,
      :sname => (ssl ? 'https' : 'http'), :info => fp)
  rescue ::Timeout::Error, ::Errno::EPIPE
  ensure
