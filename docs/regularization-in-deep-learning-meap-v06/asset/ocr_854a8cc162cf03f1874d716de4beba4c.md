# Listing 5.1 Download image data from the web link

```
import torch #A
import torch.nn.functional as F
import torchvision
from torchvision import datasets, transforms
from torch import nn
import matplotlib.pyplot as plt
import numpy as np
from PIL import Image
import requests
from io import BytesIO
torch.manual_seed(8) #B
url='https://raw.githubusercontent.com/jackliu333/regularization_in_deep_learning/main/ImageNet.png' #C
response = requests.get(url) #C
img = Image.open(BytesIO(response.content)) #D
img = np.array(img) #E
img = img[..., :3] #F
```

#A Import supporting libraries
#B set seed to ensure reproducibility
#C initiate an HTTP request via the get method
#D access the content of the response to retrieve the image
#E convert to numpy array
#F ignore the fourth channel since it's not needed

We can then plot the image as follows, whose output is shown in figure 5.3.

```
>>> plt.imshow(img, interpolation='nearest')
>>> plt.show()
```
