The second conceptual header file Shape.h proves to be much more interesting:

```
//---- <Shape.h> ---------------------

enum ShapeType ①
{
    circle,
    square
};

class Shape ②
{
protected:
    explicit Shape( ShapeType type )
        : type_( type ) ⑤
    {}

public:
    virtual ~Shape() = default; ③

    ShapeType getType() const { return type_; } ⑥

private:
    ShapeType type_; ④
};
```
