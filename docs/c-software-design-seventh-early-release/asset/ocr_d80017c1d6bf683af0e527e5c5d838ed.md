In order to properly implement the Strategy design pattern, you have to extract the implementation details of each shape separately. That means that you have to introduce one DrawStrategy class for each kind of shape:

```
//---- <DrawCircleStrategy.h> ----------------

class Circle;

class DrawCircleStrategy
{
public:
    virtual ~DrawCircleStrategy() = default;

    virtual void draw( Circle const& circle, /*some arguments*/ )
const = 0;
};

//---- <Circle.h> ----------------

#include <Shape.h>
#include <DrawCircleStrategy.h>
#include <memory>
#include <utility>

class Circle : public Shape
{
public:
    explicit Circle( double radius,
std::unique_ptr<DrawCircleStrategy> drawer )
        : radius_( radius )
        , drawer_( std::move(drawer) )
    {
        /* Checking that the given radius is valid and that
           the given 'std::unique_ptr' is not a nullptr */
    }

    void draw( /*some arguments*/ ) const override
    {
        drawer_->draw( *this, /*some arguments*/ );
    }

    double radius() const { return radius_; }

private:
```
