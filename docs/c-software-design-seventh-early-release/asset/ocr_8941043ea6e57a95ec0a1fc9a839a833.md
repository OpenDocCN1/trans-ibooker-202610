```
#include <cstdlib>
#include <iostream>
#include <span>
#include <vector>

void print( std::span<int> s ) ①
{
    std::cout << " (";
    for( int i : s ) {
        std::cout << ' ' << i;
    }
    std::cout << " )\n";
}

int main()
{
    std::vector<int> v{ 1, 2, 3, 4 };  ②

    std::vector<int> const w{ v };  ③
    std::span<int> const s{ v };  ④

    w[2] = 99;  // Compilation error!  ⑤
    s[2] = 99;  // Works!  ⑥

    // Prints ( 1 2 99 4 );
    print( s );  ⑦

    v = { 5, 6, 7, 8, 9 };  ⑧
    s[2] = 99;  // Works!  ⑨

    // Prints ?
    print( s );  ⑩

    return EXIT_SUCCESS;
}
```

The print() function (①) demonstrates the purpose of std::span. The std::span class template represents an abstraction for an array. The print() function can be called with any kind of array (built-in arrays, std::array, std::vector, ...) without coupling to any specific type of array.
